# requirements.txt
# =============================================================================
# Python dependencies for the chess book-exit research pipeline.
# Install with:
#
#   pip install -r requirements.txt
#
# Version policy:
#   Pure-Python packages are pinned exactly (==) for reproducibility.
#   Packages with C extensions use >= lower bounds so pip can find a
#   pre-built wheel for the installed Python version and OS. Without pre-built
#   wheels, pip tries to compile from C source, which requires a C
#   compiler that Windows does not ship by default.
# =============================================================================

# ── Chess game parsing and engine interaction ─────────────────────────────────
chess==1.10.0           # Pure Python; pinned exactly. PGN parsing, board
                        # representation, Stockfish integration via chess.engine.

# ── Data compression ──────────────────────────────────────────────────────────
zstandard>=0.23.0       # C extension. 0.23.0+ has pre-built Windows wheels for
                        # Python 3.12 and 3.13. Lichess .pgn.zst files use
                        # Zstandard compression, which gzip tools cannot read.

# ── Networking ────────────────────────────────────────────────────────────────
requests==2.31.0        # Pure Python; pinned exactly. HTTP downloads with
                        # resume support via Range headers.

# ── Progress display ──────────────────────────────────────────────────────────
tqdm==4.66.0            # Pure Python; pinned exactly. Progress bars for
                        # long-running loops.

# ── Data handling ─────────────────────────────────────────────────────────────
pandas>=2.2.2           # C extension. 2.2.2+ has Python 3.13 wheels.
numpy>=2.0.0            # C extension. 2.0+ has Python 3.13 wheels.

# ── Statistics ────────────────────────────────────────────────────────────────
statsmodels>=0.14.2     # C extension. 0.14.2+ has Python 3.13 wheels.
scipy>=1.13.0           # C extension. 1.13.0+ has Python 3.13 wheels.
scikit-posthocs==0.8.0  # Pure Python; pinned exactly. Dunn's post-hoc test.

# ── Figures ───────────────────────────────────────────────────────────────────
matplotlib>=3.9.0       # C extension. 3.9.0+ has Python 3.13 wheels.
seaborn>=0.13.2         # Pure Python; lower bound only.

# ── Testing ───────────────────────────────────────────────────────────────────
pytest==8.0.0           # Pure Python; pinned exactly.

# ── Notebooks (optional, not included) ────────────────────────────────────────
# jupyter is intentionally excluded. On Windows, Jupyter's labextensions
# create file paths longer than the 260-character default limit, which causes
# an OSError during installation unless Windows long-path support is enabled.
# None of the pipeline scripts use Jupyter, so it is not needed to run or
# reproduce this analysis.
